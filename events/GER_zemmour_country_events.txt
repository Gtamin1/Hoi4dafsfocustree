# 8 — Referendum Night.
country_event = {
	id = GER_zem_country.8
	title = GER_zem_country.8.t
	desc = GER_zem_country.8.d
	picture = GFX_report_event_zem_referendum_night

	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			if = {
				limit = { neutrality > 0.40 }
				set_country_flag = GER_zem_referendum_won
			}
			else_if = {
				limit = { neutrality > 0.25 }
				set_country_flag = GER_zem_referendum_partial
			}
		}
	}

	option = {
		name = GER_zem_country.8.a
		trigger = { has_country_flag = GER_zem_referendum_won }
		add_stability = 0.05
		add_popularity = { ideology = neutrality popularity = 0.05 }
	}
	option = {
		name = GER_zem_country.8.b
		trigger = { has_country_flag = GER_zem_referendum_partial }
		add_stability = 0.02
		add_popularity = { ideology = neutrality popularity = 0.02 }
	}
	option = {
		name = GER_zem_country.8.c
		trigger = {
			NOT = { has_country_flag = GER_zem_referendum_won }
			NOT = { has_country_flag = GER_zem_referendum_partial }
		}
		add_stability = -0.02
		add_political_power = 50
	}
}
